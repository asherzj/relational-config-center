//go:build integration

package main

import (
	"encoding/json"
	"os"
	"regexp"
	"testing"
)

// Temporary build artifact generator; formal migration validation is performed
// separately after this real MySQL schema has been captured.
func TestGenerateAssociationManifest(t *testing.T) {
	_, driver := startIntegrationMySQL(t)
	requireSchemaMigrationState(t, buildSchemaMigrationReleaseAt(t, 8), driver, "current", "up")
	config := *driver
	config.MultiStatements = true
	db := deliveryDB(t, &config)
	root := "../../internal/infrastructure/mysql/migrations/"
	ddl, err := os.ReadFile(root + "00009_table_release_templates.sql")
	if err != nil { t.Fatal(err) }
	if _, err := db.Exec(string(ddl)); err != nil { t.Fatal(err) }
	old, err := os.ReadFile(root + "00008_schema.json")
	if err != nil { t.Fatal(err) }
	var manifest map[string]string
	if err := json.Unmarshal(old, &manifest); err != nil { t.Fatal(err) }
	manifest["rcc_table_release_templates"] = ""
	for name := range manifest {
		var table, definition string
		if err := db.QueryRow("SHOW CREATE TABLE `"+name+"`").Scan(&table, &definition); err != nil { t.Fatal(err) }
		manifest[name] = regexp.MustCompile(` AUTO_INCREMENT=[0-9]+`).ReplaceAllString(definition, "")
	}
	encoded, err := json.MarshalIndent(manifest, "", "  ")
	if err != nil { t.Fatal(err) }
	if err := os.WriteFile(root+"00009_schema.json", append(encoded, '\n'), 0644); err != nil { t.Fatal(err) }
}
